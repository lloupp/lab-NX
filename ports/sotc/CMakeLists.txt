cmake_minimum_required(VERSION 3.16)
project(sotc_iso_bridge LANGUAGES CXX)
add_library(sotc_iso_bridge STATIC iso_disc.cpp)
target_compile_features(sotc_iso_bridge PUBLIC cxx_std_17)
target_include_directories(sotc_iso_bridge PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(sotc_iso_bridge PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(sotc_iso_probe probe.cpp)
target_link_libraries(sotc_iso_probe PRIVATE sotc_iso_bridge)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_sotc_iso_disc ../../tests/native/test_sotc_iso_disc.cpp)
  target_link_libraries(test_sotc_iso_disc PRIVATE sotc_iso_bridge)
  add_test(NAME sotc_iso_disc COMMAND test_sotc_iso_disc)
endif()
